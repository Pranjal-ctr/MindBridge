# Kio Backend — runtime dependencies
#
# Exact pins. A deploy must install the same tree that was tested; `.*` ranges
# let a patch release land between CI and production with nothing to diff.
# Test-only tools live in requirements-dev.txt so they stay out of the image.
#
# To upgrade: bump the pin here, run the suite, commit the new pin.

uvicorn[standard]==0.34.3
python-multipart==0.0.32

# --- Database ------------------------------------------------------
sqlalchemy[asyncio]==2.0.50
asyncpg==0.30.0
alembic==1.14.1
greenlet==3.5.1          # required by SQLAlchemy's asyncio bridge

# --- Validation / settings ------------------------------------------
pydantic==2.13.4

email-validator==2.3.0

# --- Auth ----------------------------------------------------------
python-jose[cryptography]==3.3.0
cryptography==49.0.0
passlib[bcrypt]==1.7.4
# passlib 1.7.4 reads bcrypt.__about__, removed in bcrypt 4.1.x+. 4.1.3 is the
# last release that still exposes it, so this pin is load-bearing: bumping it
# makes passlib log a trapped version error on every hash.
bcrypt==4.1.3

# --- HTTP client (Resend email provider, Google token verification) --
httpx==0.28.1

# --- AI ------------------------------------------------------------
google-genai==2.10.0
# Imported directly by app/auth/google.py for Google ID-token verification.
# Declared explicitly rather than relied on as a google-genai transitive.
google-auth==2.55.0

# --- Timezone database ----------------------------------------------
# The availability engine resolves IANA zones (Asia/Kolkata) via zoneinfo.
# Windows ships no system tz database at all, and slim container images
# frequently omit it, so zoneinfo falls back to this package. Without it
# ZoneInfo() raises and every counselor's availability is unresolvable.
tzdata==2025.2
